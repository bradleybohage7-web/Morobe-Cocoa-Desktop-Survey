  <script>
    // PASTE YOUR GOOGLE APPS SCRIPT WEB APP URL HERE:
    const SCRIPT_URL = "YOUR_GOOGLE_APPS_SCRIPT_WEB_APP_URL";

    // Initialize Map centered on Morobe Province
    const map = L.map('map').setView([-6.72, 146.99], 9);
    L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
      attribution: '© OpenStreetMap'
    }).addTo(map);

    let marker;
    let boundaryData = null;

    // Load the converted Shapefile (boundaries.geojson)
    fetch('boundaries.geojson')
      .then(response => response.json())
      .then(data => {
        boundaryData = data;
        L.geoJSON(data, { style: { color: "#2c5e1a", weight: 1.5, fillOpacity: 0.05 } }).addTo(map);
      })
      .catch(err => console.log("Boundary file not loaded yet:", err));

    function getLocation() {
      const status = document.getElementById('gpsStatus');
      if (!navigator.geolocation) {
        status.textContent = "Geolocation is not supported by your browser.";
        return;
      }
      status.textContent = "Acquiring high-accuracy GPS satellites...";

      navigator.geolocation.getCurrentPosition(
        (position) => {
          const lat = position.coords.latitude;
          const lon = position.coords.longitude;
          const acc = position.coords.accuracy.toFixed(1);

          document.getElementById('lat').value = lat.toFixed(6);
          document.getElementById('lon').value = lon.toFixed(6);
          document.getElementById('accuracy').value = acc + " m";
          status.textContent = `GPS Locked! Accuracy: ±${acc} meters`;

          // Update Map
          map.setView([lat, lon], 15);
          if (marker) map.removeLayer(marker);
          marker = L.marker([lat, lon]).addTo(map);

          // Spatial Check: Find which District/LLG polygon contains this point
          detectBoundary(lon, lat);
        },
        (error) => {
          status.textContent = "Error getting GPS: " + error.message;
        },
        { enableHighAccuracy: true, timeout: 15000, maximumAge: 0 }
      );
    }

    function detectBoundary(lon, lat) {
      if (!boundaryData) return;
      const pt = turf.point([lon, lat]);

      for (let feature of boundaryData.features) {
        if (turf.booleanPointInPolygon(pt, feature)) {
          // UPDATE THESE PROPERTY NAMES to match your Shapefile's exact column names!
          const dist = feature.properties.DIST_NAME || feature.properties.District || "";
          const llg = feature.properties.LLG_NAME || feature.properties.LLG || "";
          
          document.getElementById('district').value = dist;
          document.getElementById('llg').value = llg;
          break;
        }
      }
    }

    function submitData(e) {
      e.preventDefault();
      const btn = document.getElementById('submitBtn');
      btn.disabled = true;
      btn.textContent = "Uploading...";

      const payload = {
        lat: document.getElementById('lat').value,
        lon: document.getElementById('lon').value,
        accuracy: document.getElementById('accuracy').value,
        district: document.getElementById('district').value,
        llg: document.getElementById('llg').value,
        village: document.getElementById('village').value,
        ownerName: document.getElementById('ownerName').value,
        phone: document.getElementById('phone').value,
        status: document.getElementById('status').value,
        notes: document.getElementById('notes').value
      };

      fetch(SCRIPT_URL, {
        method: "POST",
        body: JSON.stringify(payload)
      })
      .then(res => res.json())
      .then(response => {
        alert("Fermentary data saved successfully!");
        document.getElementById('surveyForm').reset();
        btn.disabled = false;
        btn.textContent = "☁️ Save & Upload Record";
      })
      .catch(err => {
        alert("Upload failed. Check internet connection.");
        btn.disabled = false;
        btn.textContent = "☁️ Save & Upload Record";
      });
    }
  </script>
